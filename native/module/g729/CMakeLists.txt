if(NOT BCG729_INCLUDE_DIR OR NOT BCG729_LIBRARY)
  message(FATAL_ERROR "g729 needs BCG729_INCLUDE_DIR and BCG729_LIBRARY (scripts/build-native.mjs sets them).")
endif()

  add_library(g729 OBJECT g729.c)
else()
  add_library(g729 MODULE g729.c)
endif()

target_compile_definitions(g729 PRIVATE BCG729_STATIC)
target_link_libraries(g729 PRIVATE ${BCG729_LIBRARY})
